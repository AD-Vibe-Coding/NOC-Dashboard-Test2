// api/audit/analyze.ts

import { Anthropic } from "@anthropic-ai/sdk";
import { SYSTEM_PROMPT } from "./prompts"; // Your condensed prompt
import { parseAuditJson } from "@/src/widgets/TicketAudit/data";

const client = new Anthropic();

export interface TicketInput {
  ticket_number: string;
  owner: string;
  contributors?: string[];
  priority: string;
  service_type: string;
  issue_summary: string;
  full_ticket_content: string;
  // ... other ticket fields
}

export async function analyzeTicket(ticketData: TicketInput) {
  try {
    // Call LLM with condensed prompt
    const response = await client.messages.create({
      model: "claude-3-5-sonnet-20241022",
      max_tokens: 4000,
      system: SYSTEM_PROMPT,
      messages: [
        {
          role: "user",
          content: `Please audit the following ticket:\n\n${formatTicketForAudit(ticketData)}`
        }
      ]
    });

    // Extract JSON from response
    const auditJson = response.content[0].type === "text" 
      ? response.content[0].text 
      : "";

    // Parse and validate
    const auditResult = parseAuditJson(auditJson);

    // Return to application only (NO Sheets export)
    return {
      success: true,
      data: auditResult,
      timestamp: new Date().toISOString()
    };
  } catch (error) {
    return {
      success: false,
      error: error instanceof Error ? error.message : "Unknown error",
      timestamp: new Date().toISOString()
    };
  }
}

function formatTicketForAudit(ticketData: TicketInput): string {
  return `
Ticket #: ${ticketData.ticket_number}
Owner: ${ticketData.owner}
Contributors: ${ticketData.contributors?.join(", ") || "None"}
Priority: ${ticketData.priority}
Service Type: ${ticketData.service_type}

Issue Summary:
${ticketData.issue_summary}

Full Ticket Content:
${ticketData.full_ticket_content}
  `;
}
